title: Async Logic (createAsyncThunk)
description: Fetch data inside Redux with thunks and handle pending, fulfilled and rejected states.
---

Reducers must be **pure and synchronous**, so API calls can't go in them. Redux handles async work with **middleware**. Redux Toolkit includes the **thunk** middleware and the `createAsyncThunk` helper.

## How it works

`createAsyncThunk(type, payloadCreator)` creates an action you can dispatch. It runs your async function and automatically dispatches three lifecycle actions:

| Action | Dispatched when |
| --- | --- |
| `thunk.pending` | The request starts |
| `thunk.fulfilled` | The promise resolves. `action.payload` is the returned value |
| `thunk.rejected` | The promise rejects. `action.error` holds the error |

You handle these in the slice's `extraReducers`.

## Example: weather for a country

<Steps>
<Step>

### Create the thunk and slice

```js title="slices/countrySlice.js"
import { createSlice, createAsyncThunk } from '@reduxjs/toolkit';
import axios from 'axios';

const initialState = {
  countryData: null,
  weatherData: null,
  status: 'idle', // 'idle' | 'loading' | 'succeeded' | 'failed'
  error: null,
};

// the payload creator receives ONE argument, so pass several values as an object
export const getWeatherData = createAsyncThunk(
  'country/getCountryWeather',
  async ({ lat, lng, country }) => {
    const response = await axios.get('https://api.openweathermap.org/data/2.5/weather', {
      params: { lat, lon: lng, appid: import.meta.env.VITE_WEATHER_KEY, units: 'metric' },
    });
    return { weatherData: response.data, countryData: country }; // → action.payload
  }
);

const countrySlice = createSlice({
  name: 'country',
  initialState,
  reducers: {
    removeCountry(state) {
      state.countryData = null;
      state.weatherData = null;
    },
  },
  // handle the actions generated by the thunk
  extraReducers: (builder) => {
    builder
      .addCase(getWeatherData.pending, (state) => {
        state.status = 'loading';
      })
      .addCase(getWeatherData.fulfilled, (state, action) => {
        state.status = 'succeeded';
        state.weatherData = action.payload.weatherData;
        state.countryData = action.payload.countryData;
      })
      .addCase(getWeatherData.rejected, (state, action) => {
        state.status = 'failed';
        state.error = action.error.message;
      });
  },
});

export const { removeCountry } = countrySlice.actions;
export default countrySlice.reducer;
```

</Step>
<Step>

### Dispatch it from a component

```jsx title="components/Card.jsx"
import { Link } from 'react-router-dom';
import { useDispatch } from 'react-redux';
import { getWeatherData } from '../slices/countrySlice';

const Card = ({ country }) => {
  const dispatch = useDispatch();

  const selectCountry = () => {
    // only one argument is allowed, so pass an object
    dispatch(
      getWeatherData({
        lat: country.latlng[0],
        lng: country.latlng[1],
        country,
      })
    );
  };

  return (
    <Link to="/country" onClick={selectCountry}>
      <div className="card">
        <h3>{country.name.common}</h3>
        <img src={country.flags.png} alt={country.name.common} />
        <p>Capital: {country.capital}</p>
        <p>Region: {country.region}</p>
      </div>
    </Link>
  );
};

export default Card;
```

</Step>
<Step>

### Read the status

```jsx
const { weatherData, status, error } = useSelector((state) => state.country);

if (status === 'loading') return <p>Loading weather...</p>;
if (status === 'failed') return <p>Error: {error}</p>;
return <p>{weatherData?.main.temp} °C</p>;
```

</Step>
</Steps>

## Useful extras

```js
// unwrap: get the payload, or throw the error, in the component
try {
  const result = await dispatch(getWeatherData(args)).unwrap();
} catch (err) {
  /* handle */
}

// rejectWithValue: return a custom error payload
createAsyncThunk('x/y', async (arg, { rejectWithValue }) => {
  try {
    return (await api.get('/x')).data;
  } catch (err) {
    return rejectWithValue(err.response.data);
  }
});
```

<Callout type="warn">
  Never commit API keys in source code. Use environment variables (`.env.local` with `VITE_` prefixed names in Vite), and keep secret keys on a server. Anything shipped to the browser is public.
</Callout>
