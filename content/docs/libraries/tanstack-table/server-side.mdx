Client-side features work for a few thousand rows. For larger datasets, let the **server** paginate, sort and filter, and have the table only **display** the current page.

## Server-side pagination and sorting with TanStack Query

```tsx
import { keepPreviousData, useQuery } from '@tanstack/react-query';
import {
  createColumnHelper,
  flexRender,
  rowPaginationFeature,
  rowSortingFeature,
  tableFeatures,
  useTable,
  type PaginationState,
  type SortingState,
} from '@tanstack/react-table';

// no row models: the server already returns the sorted page
const features = tableFeatures({ rowSortingFeature, rowPaginationFeature });

type Order = { id: number; customer: string; total: number; createdAt: string };
const columnHelper = createColumnHelper<typeof features, Order>();
const columns = columnHelper.columns([
  columnHelper.accessor('id', { header: '#' }),
  columnHelper.accessor('customer', { header: 'Customer' }),
  columnHelper.accessor('total', { header: 'Total' }),
  columnHelper.accessor('createdAt', { header: 'Date' }),
]);

type OrdersResponse = { rows: Order[]; totalCount: number };

async function fetchOrders(pagination: PaginationState, sorting: SortingState): Promise<OrdersResponse> {
  const params = new URLSearchParams({
    page: String(pagination.pageIndex + 1),
    limit: String(pagination.pageSize),
  });
  if (sorting[0]) {
    params.set('sortBy', sorting[0].id);
    params.set('order', sorting[0].desc ? 'desc' : 'asc');
  }
  const res = await fetch(`/api/orders?${params}`);
  if (!res.ok) throw new Error('Failed to load orders');
  return res.json();
}

const EMPTY: Order[] = []; // stable fallback reference

export function OrdersTable() {
  const [pagination, setPagination] = useState<PaginationState>({ pageIndex: 0, pageSize: 20 });
  const [sorting, setSorting] = useState<SortingState>([]);

  // table state lives in the query key, so any change triggers a refetch
  const { data, isFetching, isPlaceholderData } = useQuery({
    queryKey: ['orders', pagination, sorting],
    queryFn: () => fetchOrders(pagination, sorting),
    placeholderData: keepPreviousData, // keep the old page visible while loading
  });

  const table = useTable({
    features,
    columns,
    data: data?.rows ?? EMPTY,
    rowCount: data?.totalCount ?? 0,   // lets the table work out the page count
    manualPagination: true,            // the server paginates
    manualSorting: true,               // the server sorts
    state: { pagination, sorting },
    onPaginationChange: setPagination,
    onSortingChange: (updater) => {
      setSorting(updater);
      setPagination((p) => ({ ...p, pageIndex: 0 })); // back to page 1 on re-sort
    },
  });

  return (
    <div style={{ opacity: isPlaceholderData ? 0.6 : 1 }}>
      <table>{/* render headers and rows exactly as in the basic example */}</table>
      <button onClick={() => table.previousPage()} disabled={!table.getCanPreviousPage()}>Prev</button>
      <span>
        {pagination.pageIndex + 1} / {table.getPageCount()}
      </span>
      <button onClick={() => table.nextPage()} disabled={!table.getCanNextPage()}>Next</button>
      {isFetching && <small> Loading...</small>}
    </div>
  );
}
```

| Option | Effect |
| --- | --- |
| `manualPagination` | Don't slice rows; `data` is already one page |
| `manualSorting` | Don't sort; the server returns sorted data |
| `manualFiltering` | Don't filter; send filter state to the server |
| `rowCount` (or `pageCount`) | The total, so `getPageCount()` and `getCanNextPage()` work |

<Callout title="Debounce filters">
  When filters go to the server, debounce the text input (see [`useDebounce`](/docs/hooks/custom-hooks#usedebounce)) before putting the value into the query key.

## Virtualising large tables

Rendering 10,000 `<tr>` elements makes the page slow. **TanStack Virtual** renders only the rows inside the viewport, plus a few extra (the overscan).

```tsx
import { useRef } from 'react';
import { useVirtualizer } from '@tanstack/react-virtual';

function VirtualTable({ table }) {
  const { rows } = table.getRowModel();
  const scrollRef = useRef<HTMLDivElement>(null);

  const virtualizer = useVirtualizer({
    count: rows.length,
    getScrollElement: () => scrollRef.current,
    estimateSize: () => 36, // row height in px
    overscan: 10,
  });

  return (
    <div ref={scrollRef} style={{ height: 600, overflow: 'auto' }}>
      {/* a full-height spacer so the scrollbar is the right size */}
      <div style={{ height: virtualizer.getTotalSize(), position: 'relative' }}>
        {virtualizer.getVirtualItems().map((virtualRow) => {
          const row = rows[virtualRow.index];
          return (
            <div
              key={row.id}
              style={{
                position: 'absolute',
                top: 0,
                left: 0,
                width: '100%',
                height: virtualRow.size,
                transform: `translateY(${virtualRow.start}px)`,
                display: 'flex',
              }}
            >
              {row.getVisibleCells().map((cell) => (
                <div key={cell.id} style={{ flex: 1 }}>
                  {flexRender(cell.column.columnDef.cell, cell.getContext())}
                </div>
              ))}
            </div>
          );
        })}
      </div>
    </div>
  );
}
```

TanStack Virtual works for any long list (chat messages, feeds, dropdowns with thousands of options), not just tables. Combine it with [`useInfiniteQuery`](/docs/libraries/tanstack-query/pagination-and-infinite) for infinite scrolling.

| Rows | Approach |
| --- | --- |
| up to ~1,000 | Client-side features; render normally |
| 1,000–50,000 already loaded | Client-side features + **virtualisation** |
| More, or expensive to load | **Server-side** pagination, sorting and filtering |
